# 1Heath business model v6 (24 Nov 2026), after the second outside review (Gemini and ChatGPT), checked at source.
# Changes from v5: medicine sales counted in full as revenue when 1Heath owns the stock (Ind AS 115, principal);
# money tied up in stock counted on doctors at year end, at 60 to 75 days (Entero ran 68 days in FY26);
# lower inputs: share of doctors paying, clinics with a counter that order, monthly buying, shares, fee per order,
# labs; 7,500 scribed visits a doctor a year in the plan case; higher pay for clinical and senior staff; a line for
# medical-device software compliance (CDSCO guidance of July 2026); a slower ramp to 10,000 doctors in year 5.
CR=1e7; L=1e5; GST=1.18; PAYFEE=0.0236
# Two cases. 'founder': what the founder heard from doctors in the field. 'cautious': the outside reviewers' corrections.
cases={
 'founder': dict(price=299,pay=0.7,clinics=0.7,buy=150000,p_share=0.60,orders=8,fee=100,own_share=0.35,gross=0.13,dist=0.10,wc=60,labs=40,lab_m=15000,ai=1.50,visits=9000,wa=51),
 'cautious':dict(price=299,pay=0.6,clinics=0.5,buy=100000,p_share=0.50,orders=6,fee=75,own_share=0.25,gross=0.13,dist=0.10,wc=60,labs=30,lab_m=15000,ai=1.50,visits=7500,wa=51),
}
RAMP={'founder':[100,1000,4000,10000,20000],'cautious':[100,1000,3000,6000,10000]}
people={'Engineering, AI and data':([6,12,18,26,34],20),'Clinical team (doctors, pharmacist, safety lead)':([2,3,5,6,8],35),
        'Field sales and clinic onboarding':([1,6,12,20,30],6),'Support for doctors and pharmacies':([1,4,8,14,22],4.5),
        'Management, finance and admin':([1,2,3,5,6],40)}
other={'Cloud servers and storage':[10,25,50,90,140],'Security, privacy, medical-device software compliance':[20,60,70,80,90],
       'Hardware':[10,10,15,20,25],'Software and tools':[5,10,18,28,40],'Legal and accounts':[20,30,40,50,60],
       'Insurance':[3,8,12,18,25],'Office':[6,15,25,40,60],'Launch, events, institution visits, marketing':[15,60,100,150,200]}
def fixed(y): return sum(n[y-1]*c for n,c in people.values())*L + sum(v[y-1] for v in other.values())*L
def year(c,route,d=1000,ai=None):
    ai=c['ai'] if ai is None else ai
    subs=d*c['pay']*c['price']/GST*12; lab=c['labs']*d/1000*c['lab_m']*12
    if route=='own':
        sales=d*c['clinics']*c['buy']*c['own_share']*12; gp=sales*c['gross']; dist=sales*c['dist']; fees=0
    else:
        sales=d*c['clinics']*c['buy']*c['p_share']*12; gp=0; dist=0; fees=d*c['clinics']*c['orders']*12*c['fee']
    rev_full=subs+lab+fees+(sales if route=='own' else 0)        # revenue as reported (Ind AS 115)
    contrib_in=subs+lab+fees+gp                                    # what 1Heath keeps before its own costs
    aic=d*c['visits']*ai; wa=d*c['wa']*12; pay=d*c['pay']*c['price']*12*PAYFEE
    var=dist+aic+wa+pay
    return dict(subs=subs,lab=lab,fees=fees,sales=sales,gp=gp,rev=rev_full,keep=contrib_in,dist=dist,ai=aic,wapay=wa+pay,var=var)
def wc(c,d): return d*c['clinics']*c['buy']*c['own_share']*12/365*c['wc']
if __name__=='__main__':
    for y in range(1,6): print(f'fixed year {y}: {fixed(y)/CR:.2f} cr; people {sum(n[y-1] for n,_ in people.values())}')
    for k,(n,c) in people.items(): print(f'  {k}: {n[1]} × ₹{c}L = {n[1]*c/100:.2f}')
    for k,v in other.items(): print(f'  {k}: {v[1]/100:.2f}')
    F=fixed(2)
    for n,c in cases.items():
        for r in ('partner','own'):
            y=year(c,r); res=y['keep']-y['var']-F; con=(y['keep']-y['var'])/1000
            print(f'\n{n}/{r}: '+', '.join(f'{k} {v/CR:.2f}' for k,v in y.items())+f' | fixed {F/CR:.2f} result {res/CR:+.2f} | per doctor/mo ₹{con/12:,.0f} | breakeven {F/con if con>0 else float("nan"):,.0f} | wc {wc(c,1000)/CR:.2f}')
    for n,c in cases.items():
        moved=c['buy']*c['p_share']; print(f'{n}: ordered ₹{moved:,.0f}/mo, 5% saving ₹{moved*0.05:,.0f}')
    def five(case,ai_path,own_from=4,label=''):
        ends=RAMP[case]; starts=[0]+ends[:-1]
        tot=0; worst=0; print('\nfive-year',case,label)
        for i in range(1,6):
            c=cases[case]; r='own' if i>=own_from else 'partner'; a=(starts[i-1]+ends[i-1])/2
            y=year(c,r,d=a,ai=ai_path[i-1])
            keep,var,rev=(0,y['ai']+y['wapay'],0) if i==1 else (y['keep'],y['var'],y['rev'])
            res=keep-var-fixed(i); tot+=res; worst=min(worst,tot)
            w=wc(c,ends[i-1]) if r=='own' else 0
            print(f' Y{i} {r:7s} end {ends[i-1]:>6,} revenue {rev/CR:7.2f} kept {keep/CR:6.2f} var {var/CR:6.2f} fixed {fixed(i)/CR:6.2f} result {res/CR:+6.2f} cum {tot/CR:+7.2f} wc_end {w/CR:6.2f}')
        print(f' cumulative {tot/CR:+.2f}; deepest {worst/CR:+.2f}')
    for cs in cases:
        five(cs,[1.5,1.5,1.0,0.75,0.5],label='own stock from year 4, AI falls')
        five(cs,[1.5]*5,label='AI flat')
        five(cs,[1.5,1.5,1.0,0.75,0.5],own_from=9,label='partner only')
    c=cases['founder']
    def r(route,**k): y=year(dict(c,**k),route); return (y['keep']-y['var']-fixed(2))/CR
    b={x:r(x) for x in ('partner','own')}
    print('\nlevers')
    for route,k,v in [('own','gross',0.16),('own','gross',0.10),('own','own_share',0.40),('partner','fee',150),('partner','ai',1.0),('partner','price',499),('partner','price',199),('own','dist',0.12),('partner','visits',9000),('partner','labs',40)]:
        print(f' {route} {k}={v}: {r(route,**{k:v})-b[route]:+.2f}')
